\section{Tipos de ondas}

\begin{enumerate}[label=\textbf{1.\arabic*.}]

\item Considere a tensão
\[
v(t)=170\sin(100\pi t)\ \text{V}.
\]
Determine:
\begin{enumerate}[label=\alph*)]
    \item amplitude;
    \item frequência angular;
    \item frequência;
    \item período;
    \item valor eficaz.
\end{enumerate}

\item O gráfico a seguir representa uma tensão senoidal.

\begin{enumerate}[label=\alph*)]
    \item valor de pico;
    \item valor pico a pico;
    \item período;
    \item frequência;
    \item valor eficaz.
\end{enumerate}

\item As três formas de onda abaixo possuem amplitude de pico igual a \SI{10}{V}.

Determine a defasagem entre as ondas e indique qual delas está adiantada.

\item Um sinal retangular unipolar assume \SI{10}{V} durante 25\% do período e \SI{0}{V} durante o restante. Determine:
\begin{enumerate}[label=\alph*)]
    \item valor médio;
    \item valor eficaz.
\end{enumerate}

\item Uma tensão contém uma componente fundamental de \SI{10}{V} RMS e uma terceira harmônica de \SI{4}{V} RMS. Desprezando outras componentes, determine:
\begin{enumerate}[label=\alph*)]
    \item valor eficaz total;
    \item THD.
\end{enumerate}

\end{enumerate}

\gabarito

\begin{enumerate}[label=\textbf{1.\arabic*.}]
\item $V_p=170$ V; $\omega=100\pi$ rad/s; $f=50$ Hz; $T=20$ ms; $V_{\rm RMS}\approx120{,}2$ V.
\item $V_p=10$ V; $V_{pp}=20$ V; $T=20$ ms; $f=50$ Hz; $V_{\rm RMS}=10/\sqrt2\approx7{,}07$ V.
\item Senoidal: $7{,}07$ V; quadrada: $10$ V; triangular: $10/\sqrt3\approx5{,}77$ V.
\item A defasagem é $45\graus$. $v_2(t)$ está adiantada em relação a $v_1(t)$.
\item $V_{\rm med}=2{,}5$ V; $V_{\rm RMS}=5$ V.
\item $V_{\rm RMS}=\sqrt{10^2+4^2}\approx10{,}77$ V; $\mathrm{THD}=4/10=40\%$.
\end{enumerate}

\begin{center}
\begin{circuitikz}

\gabarito

\begin{enumerate}[label=\textbf{3.\arabic*.}]
\item $(V_A-20)/5+V_A/(j5)=0 \Rightarrow V_A=10+j10=14{,}14\angle45^\circ$ V.
\item $10I_1-5I_2=10$ e $-5I_1+10I_2=0$; $I_1=4/3$ A e $I_2=2/3$ A.
\item $I^{(1)}=2$ A e $I^{(2)}=j1$ A; $I=2+j1=2{,}236\angle26{,}57^\circ$ A.
\item $V_{\rm Th}=20\angle0^\circ$ V; $Z_{\rm Th}=5+j5\,\Omega$; com $Z_L=5-j5\,\Omega$, $I_L=2\angle0^\circ$ A.
\item $(V_A-20)/10+(V_A-j10)/10+V_A/10=0$; $V_A=6{,}667+j3{,}333=7{,}454\angle26{,}57^\circ$ V.
\item A análise nodal tende a ser mais conveniente porque o circuito possui poucos nós essenciais e uma fonte de corrente conectada a um dos nós, o que simplifica a aplicação direta da LKC.
\end{enumerate}
